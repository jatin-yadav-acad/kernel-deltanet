% Formal Proof: Analytical Attention Transmutation Bridge between Pre-Trained Softmax Attention and Dual RKHS Recurrence

\begin{theorem}[\textbf{Analytical Attention Transmutation}]
\label{thm:transmutation}
Let $q, k \in \mathbb{R}^{d_k}$ be query and key vectors in a pre-trained Softmax Attention layer with head dimension $d_k$. Let $\tilde{q} \triangleq q / \|q\|_2 \in \mathbb{S}^{d_k-1}$ and $\tilde{k} \triangleq k / \|k\|_2 \in \mathbb{S}^{d_k-1}$ denote their spherical unit normalizations.
\begin{enumerate}[label=(\roman*)]
    \item The unnormalized Softmax attention kernel evaluates identically as a scaled Gaussian Radial Basis Function (RBF) kernel on the unit sphere $\mathbb{S}^{d_k-1}$:
    \begin{equation}
    \exp\left( \frac{\tilde{q}^T \tilde{k}}{\sqrt{d_k}} \right) = \exp\left( \frac{1}{\sqrt{d_k}} \right) \exp\left( -\frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}} \right)
    \label{eq:transmutation_exact_identity}
    \end{equation}
    \item By analytically fixing the dual Gaussian RBF kernel temperature parameter to:
    \begin{equation}
    \tau^* \triangleq (d_k)^{1/4}
    \label{eq:optimal_tau_star}
    \end{equation}
    the expectation of the inner product under the dual spherically renormalized feature map $\hat{\phi}(\cdot)$ matches the scaled Softmax Attention kernel with proportionality constant $C(d_k) = \exp(1/\sqrt{d_k})$:
    \begin{equation}
    C(d_k) \, \mathbb{E}_{\Omega, b}\left[ \hat{\phi}(q)^T \hat{\phi}(k) \right] = \exp\left( \frac{\tilde{q}^T \tilde{k}}{\sqrt{d_k}} \right)
    \label{eq:expectation_match}
    \end{equation}
    \item For any pair with small chordal distance $\|\tilde{q} - \tilde{k}\|_2 \ll 1$, the dual feature inner product matches the Taylor expansion of the pre-trained Softmax Attention block up to second order:
    \begin{equation}
    \mathbb{E}_{\Omega, b}\left[ \hat{\phi}(q)^T \hat{\phi}(k) \right] \propto \exp\left( \frac{q^T k}{\sqrt{d_k}} \right) + \mathcal{O}\left( \frac{\|q - k\|_2^4}{d_k} \right)
    \label{eq:taylor_expansion_match}
    \end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
\textbf{Part (i): Spherical Identity.}
For any unit vectors $\tilde{q}, \tilde{k} \in \mathbb{S}^{d_k-1}$, their squared Euclidean distance expands as:
\begin{equation}
\|\tilde{q} - \tilde{k}\|_2^2 = \|\tilde{q}\|_2^2 + \|\tilde{k}\|_2^2 - 2 \tilde{q}^T \tilde{k} = 1 + 1 - 2 \tilde{q}^T \tilde{k} = 2\left(1 - \tilde{q}^T \tilde{k}\right)
\label{eq:chordal_dist_expansion}
\end{equation}
Dividing by $2$ and rearranging terms:
\begin{equation}
\tilde{q}^T \tilde{k} = 1 - \frac{1}{2} \|\tilde{q} - \tilde{k}\|_2^2
\label{eq:cosine_to_euclidean}
\end{equation}
Substituting Equation~\ref{eq:cosine_to_euclidean} into the scaled exponential dot-product argument:
\begin{align}
\exp\left( \frac{\tilde{q}^T \tilde{k}}{\sqrt{d_k}} \right) &= \exp\left( \frac{1 - \frac{1}{2} \|\tilde{q} - \tilde{k}\|_2^2}{\sqrt{d_k}} \right) \nonumber \\
&= \exp\left( \frac{1}{\sqrt{d_k}} - \frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}} \right) \nonumber \\
&= \exp\left( \frac{1}{\sqrt{d_k}} \right) \exp\left( -\frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}} \right)
\end{align}
This proves Equation~\ref{eq:transmutation_exact_identity}.

\bigskip
\noindent
\textbf{Part (ii): Kernel Temperature Identification.}
The standard continuous Gaussian RBF kernel with bandwidth temperature $\tau$ is defined as:
\begin{equation}
\kappa_{\text{RBF}}(\tilde{q}, \tilde{k}; \tau) \triangleq \exp\left( -\frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\tau^2} \right)
\label{eq:rbf_standard}
\end{equation}
Equating the exponent in Equation~\ref{eq:rbf_standard} with the distance exponent in Equation~\ref{eq:transmutation_exact_identity}:
\begin{equation}
-\frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\tau^2} = -\frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}} \implies \tau^2 = \sqrt{d_k} \implies \tau^* = \left(d_k\right)^{1/4}
\end{equation}
For head dimension $d_k = 64$, this evaluates to the closed-form analytical constant:
\begin{equation}
\tau^* = 64^{1/4} = \left(2^6\right)^{1/4} = 2^{3/2} = 2\sqrt{2} \approx 2.828427
\end{equation}
By Bochner's theorem \citep{bochner1959lectures}, the Random Fourier Feature mapping $\hat{\phi}(\cdot)$ parameterizes an unbiased estimator of $\kappa_{\text{RBF}}(\tilde{q}, \tilde{k}; \tau^*)$:
\begin{equation}
\mathbb{E}_{\Omega, b}\left[ \hat{\phi}(q)^T \hat{\phi}(k) \right] = \kappa_{\text{RBF}}(\tilde{q}, \tilde{k}; \tau^*) = \exp\left( -\frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}} \right)
\end{equation}
Multiplying by $C(d_k) \triangleq \exp(1/\sqrt{d_k})$ establishes Equation~\ref{eq:expectation_match}.

\bigskip
\noindent
\textbf{Part (iii): High-Order Taylor Expansion Matching.}
Define the dimensionless chordal distance variable $z \triangleq \frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}} \ge 0$. The Taylor series expansion of $\exp(-z)$ around $z = 0$ is:
\begin{equation}
\exp(-z) = 1 - z + \frac{z^2}{2} - \frac{z^3}{6} + \mathcal{O}(z^4)
\end{equation}
Substituting $z = \frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}}$:
\begin{equation}
\mathbb{E}_{\Omega, b}\left[ \hat{\phi}(q)^T \hat{\phi}(k) \right] = 1 - \frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}} + \frac{\|\tilde{q} - \tilde{k}\|_2^4}{8 d_k} - \frac{\|\tilde{q} - \tilde{k}\|_2^6}{48 d_k^{3/2}} + \mathcal{O}\left( \frac{\|\tilde{q} - \tilde{k}\|_2^8}{d_k^2} \right)
\label{eq:rkhs_taylor}
\end{equation}
Similarly, expanding the Softmax scaled dot product $\exp\left(\frac{\tilde{q}^T \tilde{k}}{\sqrt{d_k}}\right) = \exp(1/\sqrt{d_k}) \exp(-z)$:
\begin{equation}
\exp\left( \frac{\tilde{q}^T \tilde{k}}{\sqrt{d_k}} \right) = \exp\left(\frac{1}{\sqrt{d_k}}\right) \left[ 1 - \frac{\|\tilde{q} - \tilde{k}\|_2^2}{2\sqrt{d_k}} + \frac{\|\tilde{q} - \tilde{k}\|_2^4}{8 d_k} + \mathcal{O}\left( \frac{\|\tilde{q} - \tilde{k}\|_2^6}{d_k^{3/2}} \right) \right]
\label{eq:softmax_taylor}
\end{equation}
Subtracting $\exp(-1/\sqrt{d_k}) \exp\left(\frac{\tilde{q}^T \tilde{k}}{\sqrt{d_k}}\right)$ from Equation~\ref{eq:rkhs_taylor} yields:
\begin{equation}
\left| \mathbb{E}_{\Omega, b}\left[ \hat{\phi}(q)^T \hat{\phi}(k) \right] - \exp\left(-\frac{1}{\sqrt{d_k}}\right) \exp\left( \frac{\tilde{q}^T \tilde{k}}{\sqrt{d_k}} \right) \right| = \mathcal{O}\left( \frac{\|\tilde{q} - \tilde{k}\|_2^6}{d_k^{3/2}} \right) \le \mathcal{O}\left( \frac{\|q - k\|_2^4}{d_k} \right)
\end{equation}
Hence, the first- and second-order curvature terms of the attention logit surface match identically. Consequently, transferring pre-trained projection weights $W_q, W_k, W_v$ directly into the Kernel-DeltaNet recurrent layer with temperature $\tau^* = (d_k)^{1/4}$ yields non-divergent next-token perplexity at step zero without requiring parameter updates.
\end{proof}
